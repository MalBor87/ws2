2. Čítanie poslednej hodnoty z databázy (napr. pre ESP)

<?php
require 'db.php'; // Obsahuje $server,$user, $pwd,$db

$jas = 127; // Predvolená hodnota

try {
    $conn = mysqli_connect($server,$user, $pwd,$db);
    
    // Výber posledného záznamu podľa najvyššieho ID
    $sql = "SELECT jas FROM LED ORDER BY id DESC LIMIT 1";
    $result = mysqli_query($conn,$sql);

    if ($result && $row = mysqli_fetch_assoc($result)) {
        $jas =$row['jas'];
    }

    mysqli_close($conn);
} catch (Exception $e) {
    echo "Chyba: " . $e->getMessage();
}

echo "Led=" . $jas;
?>

3. Čítanie zoznamu z DB (Prepared Statements + Výpis do HTML tabuľky)

<?php
require 'db.php';

$najdi = trim($_GET['najdi'] ?? '');$vysledky = [];

$conn = mysqli_connect($server,$user, $pwd,$db);

if (!empty($najdi)) {
    // Bezpečné vyhľadávanie pomocou Prepared Statement (LIKE)
    $sql = "SELECT id, firstname, lastname, birthday FROM Person WHERE firstname LIKE ?";
    $stmt = mysqli_prepare($conn,$sql);
    
    $searchPattern = '\%' .$najdi . '%';
    mysqli_stmt_bind_param($stmt, "s", $searchPattern);
    mysqli_stmt_execute($stmt);
    
    $result = mysqli_stmt_get_result($stmt);
    while ($row = mysqli_fetch_assoc($result)) {
        $vysledky[] =$row;
    }
    mysqli_stmt_close($stmt);
} else {
    // Načítanie všetkých záznamov
    $result = mysqli_query($conn, "SELECT id, firstname, lastname, birthday FROM Person");
    while ($row = mysqli_fetch_assoc($result)) {
        $vysledky[] =$row;
    }
}

mysqli_close($conn);
?>

<!-- Výpis načítaných dát do HTML tabuľky -->

1. ESP ako Station (STA)

#include <WiFi.h>

const char* ssid = "Tvoja_WiFi_Siet";
const char* password = "Tvoje_WiFi_Heslo";

void setup() {
  Serial.begin(115200);
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nPripojené! IP: " + WiFi.localIP().toString());
}

void loop() {}
