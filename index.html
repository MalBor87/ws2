PHP Kontrola Vstupov a Presmerovanie (`header`)

Prijatie údajov, overenie vyplnenia a bezpečné presmerovanie späť bez výstupu[cite: 1, 2]:

<?php
// Kontrola, či sú povinné vstupy vyplnené
if (empty($_GET['meno']) || empty($_GET['kraj'])) {
    // Okamžité presmerovanie s chybovým parametrom
    header('Location: index.php?chyba=1');
    exit(); // Dôležité: Zastaví ďalšie vykonávanie skriptu
}

echo "Vítaj, " . htmlspecialchars($_GET['meno']);
?>

Vkladanie dát (INSERT s `mysqli_insert_id`)

Prepojenie tabuliek `Person` a `Address` bez nutnosti manuálneho zadávania ID[cite: 1, 4]:

<?php
require 'db.php';

$meno = mysqli_real_escape_string($conn, $_POST['meno']);
$narod = !empty($_POST['narodeniny']) ? "'" . $_POST['narodeniny'] . "'" : "NULL";

// 1. Vloženie Osoby
$sql = "INSERT INTO Person (firstname, birthday) VALUES ('$meno', $narod);";
mysqli_query($conn, $sql);

// 2. Získanie automaticky vygenerovaného ID osobe
$person_id = mysqli_insert_id($conn);

// 3. Vloženie Adresy pre dané person_id
$sqlAddress = "INSERT INTO Address (person_id, street, city) VALUES ($person_id, 'Hlavná 5', 'Košice');";
mysqli_query($conn, $sqlAddress);
?>

Bezpečné vyhľadávanie cez Prepared Statements (`LIKE`)

<?php
$hladaj = '%' . $_GET['najdi'] . '%';

$sql = "SELECT id, firstname, lastname FROM Person WHERE firstname LIKE ?";
$stmt = mysqli_prepare($conn, $sql);
mysqli_stmt_bind_param($stmt, "s", $hladaj);
mysqli_stmt_execute($stmt);

$result = mysqli_stmt_get_result($stmt);
while ($row = mysqli_fetch_assoc($result)) {
    echo $row['firstname'] . " " . $row['lastname'] . "<br>";
}
mysqli_stmt_close($stmt);
?>

1. ESP ako Station (STA) – Pripojenie k domácej Wi-Fi

#include <WiFi.h> // Pre ESP8266 použi <ESP8266WiFi.h>

const char* ssid = "Moja_Domaca_WiFi";
const char* password = "HesloDoWiFi123";

void setup() {
  Serial.begin(115200);
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  Serial.print("Pripájam sa k WiFi...");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nPripojené!");
  Serial.print("IP Adresa ESP: ");
  Serial.println(WiFi.localIP());
}

void loop() {}

2. ESP ako Access Point (AP) – ESP vytvorí vlastnú sieť

#include <WiFi.h>

const char* ap_ssid = "ESP_Konfiguracia";
const char* ap_password = "heslo_ap_123"; // min 8 znakov

void setup() {
  Serial.begin(115200);
  
  // Spustenie AP režimu
  WiFi.softAP(ap_ssid, ap_password);

  IPAddress myIP = WiFi.softAPIP();
  Serial.print("Access Point spustený! IP Adresa: ");
  Serial.println(myIP);
}

void loop() {}
