1. ESP ako TCP Server

#include <WiFi.h>

WiFiServer server(8888);

void setup() {
  Serial.begin(115200);
  WiFi.mode(WIFI_AP);
  WiFi.softAP("ESP_Server", "12345678");
  server.begin();
}

void loop() {
  WiFiClient client = server.available();
  if (client) {
    while (client.connected()) {
      while (client.available() > 0) {
        char c = client.read();
        Serial.write(c);
      }
      while (Serial.available() > 0) {
        client.write(Serial.read());
      }
    }
    client.stop();
  }
}

2. ESP ako TCP Klient

#include <WiFi.h>

const char* serverIP = "192.168.X.X"; // IP adresa tvojho servera
const uint16_t serverPort = 8888;

void setup() {
  Serial.begin(115200);
  WiFi.begin("Tvoja_WiFi_Siet", "Tvoje_WiFi_Heslo");
  while (WiFi.status() != WL_CONNECTED) delay(500);

  WiFiClient client;
  if (client.connect(serverIP, serverPort)) {
    Serial.println("Pripojené k TCP serveru!");
    client.println("Ahoj zo strany ESP TCP Klienta!");
    
    while (client.connected() || client.available()) {
      if (client.available()) {
        String line = client.readStringUntil('\n');
        Serial.println("Odpoveď: " + line);
      }
    }
    client.stop();
  }
}

void loop() {}

3. ESP ako UDP Server / Receiver

#include <WiFi.h>
#include <WiFiUdp.h>

WiFiUDP udp;
const unsigned int udpPort = 4567;

void setup() {
  Serial.begin(115200);
  WiFi.begin("Tvoja_WiFi_Siet", "Tvoje_WiFi_Heslo");
  while (WiFi.status() != WL_CONNECTED) delay(500);

  udp.begin(udpPort);
}

void loop() {
  int packetSize = udp.parsePacket();
  if (packetSize) {
    char buffer[255];
    int len = udp.read(buffer, 254);
    buffer[len] = '\0';
    
    Serial.print("Prijatá UDP správa od " + udp.remoteIP().toString() + ": ");
    Serial.println(buffer);
  }
}

4. ESP ako UDP Broadcast Sender

#include <WiFi.h>
#include <WiFiUdp.h>

WiFiUDP udp;
const unsigned int udpPort = 4567;

void setup() {
  Serial.begin(115200);
  WiFi.mode(WIFI_AP);
  WiFi.softAP("ESP_UDP_AP", "12345678");
}

void loop() {
  IPAddress broadcast = WiFi.broadcastIP();
  udp.beginPacket(broadcast, udpPort);
  udp.print("My IP: ");
  udp.print(WiFi.softAPIP());
  udp.print(" | TCP port: 8888");
  udp.endPacket();

  delay(1000);
}

1. HTTP GET & POST pomocou `WiFiClient` (Manuálna požiadavka)

#include <WiFi.h>

const char* host = "tvoj-web-server.sk"; // Názov domény alebo IP
const int port = 80;

void sendGET() {
  WiFiClient client;
  if (client.connect(host, port)) {
    client.println("GET /priečinok/index.php?get=1 HTTP/1.1");
    client.print("Host: ");
    client.println(host);
    client.println("Connection: close");
    client.println();

    while (client.connected() || client.available()) {
      if (client.available()) {
        String line = client.readStringUntil('\n');
        int index = line.indexOf("Led=");
        if (index != -1) {
          int jas = line.substring(index + 4).toInt();
          Serial.println("Načítaný jas: " + String(jas));
        }
      }
    }
    client.stop();
  }
}

void sendPOST() {
  WiFiClient client;
  if (client.connect(host, port)) {
    String postData = "jas=180&status=ok";

    client.println("POST /priečinok/index.php HTTP/1.1");
    client.print("Host: ");
    client.println(host);
    client.println("Content-Type: application/x-www-form-urlencoded");
    client.print("Content-Length: ");
    client.println(postData.length());
    client.println("Connection: close");
    client.println();
    client.print(postData);

    client.stop();
  }
}

2. HTTP GET / POST pomocou knižnice `HTTPClient.h` & Basic Auth

#include <WiFi.h>
#include <HTTPClient.h>

const char* serverUrl = "http://tvoj-web-server.sk/get_action.php";

void httpGetExample() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    
    // HTTP Basic Auth autorizácia (Meno, Heslo)
    http.setAuthorization("prihlasovacie_meno", "prihlasovacie_heslo");

    int httpCode = http.GET();
    if (httpCode == 200) {
      String payload = http.getString();
      Serial.println("Odpoveď: " + payload);
    } else {
      Serial.println("Chyba HTTP: " + String(httpCode));
    }
    http.end();
  }
}

void httpPostExample() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin("http://tvoj-web-server.sk/post_action.php");
    
    http.addHeader("Content-Type", "application/x-www-form-urlencoded");
    http.setAuthorization("prihlasovacie_meno", "prihlasovacie_heslo");

    int httpCode = http.POST("jas=200&stlacene=1");
    if (httpCode > 0) {
      String response = http.getString();
      Serial.println("Odpoveď zo servera: " + response);
    }
    http.end();
  }
}
